rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAllowed() {
      return request.auth != null && (
        request.auth.token.email.lower() == 'faragm100@gmail.com' ||
        (request.auth.token.email_verified == true &&
         request.auth.token.email.lower() in get(/databases/$(database)/documents/config/members).data.emails)
      );
    }
    // Private finance is isolated from all shared household collections.
    match /privateFinance/{ownerId} {
      function ownsFinance() {
        return request.auth != null
          && request.auth.uid == 'PQzokv2oTVgcN7afESwk1b7qFKI3'
          && ownerId == request.auth.uid;
      }
      allow read, delete: if ownsFinance();
      allow create, update: if ownsFinance()
        && request.resource.data.keys().hasOnly(['schemaVersion', 'payload'])
        && request.resource.data.schemaVersion == 1
        && request.resource.data.payload is string
        && request.resource.data.payload.size() < 700000;
    }
    match /purchases/{id} {
      allow read, write: if isAllowed();
    }
    match /products/{id} {
      allow read, write: if isAllowed();
    }
    match /catalog/{id} {
      allow read, write: if isAllowed();
    }
    match /photos/{id} {
      allow read, write: if isAllowed();
    }
    match /config/{doc} {
      allow read, write: if isAllowed();
    }
  }
}
